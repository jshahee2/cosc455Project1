1//! contains the lexical analyzer for the lolcode compiler
2
3use std::process;
4
5use crate::compiler::LolCompiler;
6use crate::traits::LexicalAnalyzer;
7
8/// every valid annotation
9const ANNOTATIONS: [&str; 11] = [
10    "#HAI", "#KBYE", "#OBTW", "#TLDR", "#MAEK", "#OIC", "#GIMMEH", "#MKAY", "#NEWLINE", "#IHAZ",
11    "#ITIZ",
12];
13
14/// characters allowed in plain text
15fn is_allowed_char(c: char) -> bool {
16    c.is_ascii_alphanumeric() || ",.\":?!%=/".contains(c)
17}
18
19impl LolCompiler {
20    /// prints a lexical error and stop the program
21    fn lex_error(&self, msg: &str) -> ! {
22        eprintln!("Lexical error at line {}: {}", self.line, msg);
23        process::exit(1);
24    }
25
26    /// reads a single token
27    pub fn read_token(&mut self) -> String {
28        self.lexeme.clear();
29
30        while self.pos < self.chars.len() && self.chars[self.pos].is_whitespace() {
31            self.get_char();
32        }
33        if self.pos >= self.chars.len() {
34            return String::new();
35        }
36
37        let first = self.get_char();
38        self.add_char(first);
39
40        if first == '#' {
41            while self.pos < self.chars.len() && self.chars[self.pos].is_ascii_alphabetic() {
42                let c = self.get_char();
43                self.add_char(c);
44            }
45        } else {
46            if !is_allowed_char(first) {
47                self.lex_error(&format!("illegal character '{}'", first));
48            }
49            while self.pos < self.chars.len()
50                && !self.chars[self.pos].is_whitespace()
51                && self.chars[self.pos] != '#'
52            {
53                let c = self.get_char();
54                if !is_allowed_char(c) {
55                    self.lex_error(&format!("illegal character '{}'", c));
56                }
57                self.add_char(c);
58            }
59        }
60
61        // checks if finished token is valid
62        let lexeme = self.lexeme.clone();
63        if !self.lookup(&lexeme) {
64            self.lex_error(&format!("unknown annotation '{}'", lexeme));
65        }
66
67        // store annotations as uppercase
68        if first == '#' {
69            lexeme.to_uppercase()
70        } else {
71            lexeme
72        }
73    }
74}
75
76impl LexicalAnalyzer for LolCompiler {
77    /// returns next character and moves forward
78    fn get_char(&mut self) -> char {
79        if self.pos >= self.chars.len() {
80            eprintln!("Lexical error: tried to read past the end of the file");
81            process::exit(1);
82        }
83        let c = self.chars[self.pos];
84        self.pos += 1;
85        if c == '\n' {
86            self.line += 1;
87        }
88        c
89    }
90
91    /// adds character to current token
92    fn add_char(&mut self, c: char) {
93        self.lexeme.push(c);
94    }
95
96    /// checks if its a valid token
97    fn lookup(&self, s: &str) -> bool {
98        if s.starts_with('#') {
99            ANNOTATIONS.contains(&s.to_uppercase().as_str())
100        } else {
101            !s.is_empty() && s.chars().all(is_allowed_char)
102        }
103    }
104}
